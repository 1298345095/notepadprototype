find_package(PkgConfig REQUIRED)
pkg_check_modules(WX_GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0 gmodule-no-export-2.0)
if(WX_STATIC_SRC)
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
  set(wxBUILD_TOOLKIT gtk3 CACHE STRING "" FORCE)
  # The probe uses a native GTK editor and requires no bundled Scintilla, browser or SVG decoder.
  foreach(opt IN ITEMS wxUSE_STC wxUSE_WEBVIEW wxUSE_MEDIACTRL wxUSE_GLCANVAS wxUSE_NANOSVG)
    set(${opt} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${WX_STATIC_SRC}" wx-static EXCLUDE_FROM_ALL)
else()
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
endif()
set(wx_donor "${SCINTILLA_DONOR_SRC}")
if(NOT wx_donor)
  set(wx_donor "${SCINTILLA_SRC}")
endif()
if(NOT EXISTS "${wx_donor}/gtk/ScintillaGTK.cxx")
  message(FATAL_ERROR "Set SCINTILLA_DONOR_SRC to the Scintilla 5.6.8 source directory")
endif()
set(wx_gtk "${CMAKE_CURRENT_BINARY_DIR}/gtk-port")
file(MAKE_DIRECTORY "${wx_gtk}")
file(COPY "${wx_donor}/gtk/" DESTINATION "${wx_gtk}")
file(COPY "${wx_donor}/include/ScintillaWidget.h" DESTINATION "${wx_gtk}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS fork-gtk.patch)
execute_process(COMMAND patch --batch --forward -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/fork-gtk.patch"
  WORKING_DIRECTORY "${wx_gtk}" COMMAND_ERROR_IS_FATAL ANY)
add_library(np4_gtk_adapter STATIC "${wx_gtk}/ScintillaGTK.cxx" "${wx_gtk}/ScintillaGTKAccessible.cxx"
  "${wx_gtk}/PlatGTK.cxx" "${wx_gtk}/scintilla-marshal.c")
target_include_directories(np4_gtk_adapter PUBLIC "${wx_gtk}")
target_compile_definitions(np4_gtk_adapter PUBLIC GTK)
target_compile_options(np4_gtk_adapter PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include>" "$<$<COMPILE_LANGUAGE:CXX>:fork_editor.hpp>")
target_link_libraries(np4_gtk_adapter PUBLIC np4_core PkgConfig::WX_GTK3)
add_executable(wx-study main.cpp)
target_link_libraries(wx-study PRIVATE np4_gtk_adapter np4_study)
if(WX_STATIC_SRC)
  target_link_libraries(wx-study PRIVATE wx::core wx::base)
else()
  target_include_directories(wx-study PRIVATE ${wxWidgets_INCLUDE_DIRS})
  target_compile_definitions(wx-study PRIVATE ${wxWidgets_DEFINITIONS})
  target_compile_options(wx-study PRIVATE ${wxWidgets_CXX_FLAGS})
  target_link_libraries(wx-study PRIVATE ${wxWidgets_LIBRARIES})
endif()
