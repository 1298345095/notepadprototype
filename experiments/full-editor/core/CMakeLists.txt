cmake_minimum_required(VERSION 3.20)
project(np4_linux_core LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(NP4_FORK_DIR "${CMAKE_BINARY_DIR}/np4-fork" CACHE INTERNAL "Prepared Notepad4 Scintilla source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS prepare.py ParallelPortable.h)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py" "${NOTEPAD4_SRC}" "${NP4_FORK_DIR}" COMMAND_ERROR_IS_FATAL ANY)
file(GLOB core_sources "${NP4_FORK_DIR}/src/*.cxx" "${NP4_FORK_DIR}/lexlib/*.cxx")
add_library(np4_core STATIC ${core_sources} "${NP4_FORK_DIR}/lexers/LexJSON.cxx" "${NP4_FORK_DIR}/lexers/LexNull.cxx" app_context.cpp)
target_include_directories(np4_core PUBLIC "${NP4_FORK_DIR}/include" "${NP4_FORK_DIR}/src" "${NP4_FORK_DIR}/lexlib" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(np4_core PUBLIC cxx_std_20)
target_compile_definitions(np4_core PUBLIC NO_CXX11_REGEX)
target_compile_options(np4_core PRIVATE -ffunction-sections -fdata-sections -include cstdint)
