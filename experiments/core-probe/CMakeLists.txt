cmake_minimum_required(VERSION 3.25)
project(ProbeDLL LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(repo "${CMAKE_CURRENT_LIST_DIR}/../..")
set(NOTEPAD4_ROOT "${repo}/deps/notepad4")
add_subdirectory("${repo}/engine" np4-engine)
add_library(NP4Probe SHARED probe.cpp)
target_compile_features(NP4Probe PRIVATE cxx_std_20)
target_compile_options(NP4Probe PRIVATE /utf-8)
target_compile_definitions(NP4Probe PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601 UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(NP4Probe PRIVATE np4_engine)
