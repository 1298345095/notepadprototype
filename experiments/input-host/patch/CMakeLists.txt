cmake_minimum_required(VERSION 3.20)
project(fltk_wayland_edit_probe LANGUAGES C CXX)

if(NOT TARGET fltk::fltk)
  set(FLTK_BACKEND_WAYLAND ON CACHE BOOL "")
  set(FLTK_BACKEND_X11 OFF CACHE BOOL "")
  set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "")
  foreach(opt IN ITEMS GL TEST FLUID FLTK_OPTIONS)
    set(FLTK_BUILD_${opt} OFF CACHE BOOL "")
  endforeach()
  add_subdirectory("${FLTK_SRC}" fltk-build EXCLUDE_FROM_ALL)
endif()
set(fork "${CMAKE_CURRENT_BINARY_DIR}/fork")
execute_process(COMMAND python3 "${COCOA_PROBE}/prepare.py" "${NOTEPAD4_SRC}" "${fork}" COMMAND_ERROR_IS_FATAL ANY)
set(src)
foreach(name IN ITEMS CellBuffer CharClassify CaseConvert CaseFolder ChangeHistory Decoration Document PerLine RESearch RunStyles UndoHistory UniConversion)
  list(APPEND src "${fork}/src/${name}.cxx")
endforeach()
foreach(name IN ITEMS Accessor DefaultLexer LexAccessor LexerBase LexerModule PropSetSimple WordList)
  list(APPEND src "${fork}/lexlib/${name}.cxx")
endforeach()
add_library(fltk-study-document STATIC ${src})
target_compile_features(fltk-study-document PUBLIC cxx_std_20)
target_compile_definitions(fltk-study-document PUBLIC NDEBUG NO_CXX11_REGEX)
target_compile_options(fltk-study-document PUBLIC -ffunction-sections -fdata-sections -include cstdint)
target_include_directories(fltk-study-document PUBLIC "${COCOA_PROBE}" "${fork}/src" "${fork}/include" "${fork}/lexlib")
add_executable(fltk-edit-probe native.cxx)
add_executable(fltk-patched-pinyin pinyin.cxx)
foreach(target IN ITEMS fltk-edit-probe fltk-patched-pinyin)
  target_link_options(${target} PRIVATE -Wl,--gc-sections)
  target_link_libraries(${target} PRIVATE fltk-study-document fltk::fltk)
endforeach()
