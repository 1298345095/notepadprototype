cmake_minimum_required(VERSION 3.20)
project(np4_wayland_pinyin LANGUAGES C CXX)
if(NOT TARGET fltk::fltk)
  add_subdirectory("../wayland" fltk-probe EXCLUDE_FROM_ALL)
endif()
set(fork "${CMAKE_CURRENT_BINARY_DIR}/fork")
execute_process(COMMAND python3 "${COCOA_PROBE}/prepare.py" "${NOTEPAD4_SRC}" "${fork}" COMMAND_ERROR_IS_FATAL ANY)
set(doc_names CellBuffer CharClassify CaseConvert CaseFolder ChangeHistory Decoration Document PerLine RESearch RunStyles UndoHistory UniConversion)
set(lex_names Accessor DefaultLexer LexAccessor LexerBase LexerModule PropSetSimple WordList)
set(src observer.cxx)
foreach(name IN LISTS doc_names)
  list(APPEND src "${fork}/src/${name}.cxx")
endforeach()
foreach(name IN LISTS lex_names)
  list(APPEND src "${fork}/lexlib/${name}.cxx")
endforeach()
add_executable(real-pinyin ${src})
target_compile_features(real-pinyin PRIVATE cxx_std_20)
target_compile_definitions(real-pinyin PRIVATE NDEBUG NO_CXX11_REGEX)
target_compile_options(real-pinyin PRIVATE -ffunction-sections -fdata-sections -include cstdint)
target_include_directories(real-pinyin PRIVATE "${COCOA_PROBE}" "${fork}/src" "${fork}/include" "${fork}/lexlib")
target_link_options(real-pinyin PRIVATE -Wl,--gc-sections)
target_link_libraries(real-pinyin PRIVATE fltk::fltk)
