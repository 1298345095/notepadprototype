cmake_minimum_required(VERSION 3.20)
project(np4_fltk_wayland_probe LANGUAGES C CXX)
set(FLTK_BACKEND_WAYLAND ON CACHE BOOL "")
set(FLTK_BACKEND_X11 OFF CACHE BOOL "")
set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(FLTK_BUILD_GL OFF CACHE BOOL "")
set(FLTK_BUILD_TEST OFF CACHE BOOL "")
set(FLTK_BUILD_FLUID OFF CACHE BOOL "")
set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "")
add_subdirectory("${FLTK_SRC}" fltk-build EXCLUDE_FROM_ALL)
add_executable(flwaystudy probe.cxx)
target_compile_features(flwaystudy PRIVATE cxx_std_17)
target_link_libraries(flwaystudy PRIVATE fltk::fltk)
