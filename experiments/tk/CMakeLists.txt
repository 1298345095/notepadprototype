cmake_minimum_required(VERSION 3.25)
project(TkWin32Host LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use x64 MSVC on Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(repo "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(NOTEPAD4_ROOT "${repo}/deps/notepad4")
add_subdirectory("${repo}/engine" engine)
set_source_files_properties("${TK_WISH_RESOURCE}" PROPERTIES EXTERNAL_OBJECT TRUE)
add_executable(np4-tk WIN32 tk_host.cpp "${TK_WISH_RESOURCE}")
target_link_options(np4-tk PRIVATE /MANIFEST:NO) # wish.res already contains a manifest
target_compile_features(np4-tk PRIVATE cxx_std_20)
target_compile_options(np4-tk PRIVATE /utf-8)
target_compile_definitions(np4-tk PRIVATE STATIC_BUILD UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 WINVER=0x0601)
target_include_directories(np4-tk PRIVATE "${repo}/src" "${repo}/deps/tcl/generic" "${repo}/deps/tk/generic" "${repo}/deps/tk/xlib")
target_link_libraries(np4-tk PRIVATE np4_engine "${TK_LIB}" "${TK_STUB}" "${TCL_LIB}" "${TCL_STUB}" netapi32 userenv ws2_32 gdi32 user32 winspool shell32 ole32 oleaut32 uuid oleacc comctl32 comdlg32 imm32 uxtheme)
