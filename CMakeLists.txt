cmake_minimum_required(VERSION 3.25)
project(NotepadPrototypes LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure with the x64 MSVC toolchain on Windows")
endif()

set(NP4_GUI "fltk" CACHE STRING "Select fltk or wx")
set(NOTEPAD4_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/notepad4" CACHE PATH "Pinned Notepad4 checkout")
set(GUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/gui" CACHE PATH "Pinned GUI toolkit checkout")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(engine)

if(NP4_GUI STREQUAL "fltk")
  set(FLTK_MSVC_RUNTIME_DLL OFF CACHE BOOL "" FORCE)
  foreach(option FLTK_BUILD_SHARED_LIBS FLTK_BUILD_TEST FLTK_BUILD_EXAMPLES FLTK_BUILD_FLUID FLTK_BUILD_FORMS FLTK_BUILD_FLTK_OPTIONS FLTK_BUILD_GL)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${GUI_ROOT}" gui EXCLUDE_FROM_ALL)
  set(gui_libraries fltk::fltk shell32)
elseif(NP4_GUI STREQUAL "wx")
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
  set(wxUSE_STC OFF CACHE BOOL "" FORCE)
  set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
  foreach(option wxUSE_REGEX wxUSE_ZLIB wxUSE_LIBPNG wxUSE_LIBJPEG wxUSE_LIBTIFF wxUSE_EXPAT)
    set(${option} builtin CACHE STRING "" FORCE)
  endforeach()
  add_subdirectory("${GUI_ROOT}" gui EXCLUDE_FROM_ALL)
  set(gui_libraries wx::core wx::base)
else()
  message(FATAL_ERROR "NP4_GUI must be fltk or wx")
endif()

add_executable(np4 WIN32 "src/${NP4_GUI}_host.cpp")
set_target_properties(np4 PROPERTIES OUTPUT_NAME "np4-${NP4_GUI}-win64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
target_compile_features(np4 PRIVATE cxx_std_20)
target_compile_options(np4 PRIVATE /utf-8 /Zc:__cplusplus)
target_compile_definitions(np4 PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0601 WINVER=0x0601)
target_link_libraries(np4 PRIVATE np4_engine ${gui_libraries} comctl32)
